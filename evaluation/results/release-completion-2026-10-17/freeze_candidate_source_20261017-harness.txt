from pathlib import Path
from datetime import datetime,timezone
import subprocess,hashlib,json
root=Path.cwd()
paths=set()
for directory,pattern in [('backend','*.py'),('backend/tests/fixtures','*'),('ops','*.py'),('evaluation','*.py'),('frontend/src','*'),('frontend/public','*'),('frontend/tests','*'),('evaluation/fixtures','*.json'),('data/evaluation/prototype-v1','*')]:
 for p in (root/directory).rglob(pattern):
  if p.is_file() and '__pycache__' not in p.parts: paths.add(p.relative_to(root))
for name in ['pyproject.toml','uv.lock','alembic.ini','frontend/index.html','frontend/package.json','frontend/package-lock.json','frontend/playwright.config.ts','frontend/tsconfig.json','frontend/vite.config.ts','.env.example']:
 if (root/name).is_file(): paths.add(Path(name))
manifest={'version':'final-candidate-source-v1','frozen_at':datetime.now(timezone.utc).isoformat(),'base_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'source_state':'uncommitted_unpushed_working_tree','acceptance_status':'blocked_by_recorded_gates','scope':'Application, tests, operational/evaluation runners, fixtures, frozen evaluation data and dependency/configuration files; no secrets, uploaded originals, generated builds or mutable runtime state.','sha256':{str(p):hashlib.sha256((root/p).read_bytes()).hexdigest() for p in sorted(paths)}}
p=root/'docs/validation/final-candidate-source-2026-10-17.json'
with p.open('x') as stream: stream.write(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({'files':len(paths),'manifest_sha256':hashlib.sha256(p.read_bytes()).hexdigest()}))
