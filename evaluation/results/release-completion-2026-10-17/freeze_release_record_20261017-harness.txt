from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,shutil,subprocess
root=Path.cwd();base=root/'evaluation/results/release-completion-2026-10-17'
def read(name): return json.loads((base/name).read_text())
def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
for kind in ('unit','integration'):
 source=Path('/private/tmp/progress-final-20261017-'+kind+'.xml')
 shutil.copy2(source,base/('final-'+kind+'.xml'))
targets=json.loads((root/'docs/evaluation-targets-approved-2026-10-16.json').read_text())['targets']
metrics={'atomic_event_f1_min':'initial_atomic_event_f1','selected_link_precision_min':'initial_selected_link_precision','selected_link_coverage_min':'initial_selected_link_coverage','candidate_recall_at_8_min':'candidate_recall_at_8','endpoint_exact_accuracy_min':'initial_endpoint_field_accuracy','safe_ambiguity_handling_min':'safe_ambiguity_handling'}
gates={}
for split in ('validation','holdout'):
 summary=read(split+'-scored-v1/summary.json')
 result={name:{'target':targets[name],'observed':summary[key],'passed':summary[key] is not None and summary[key]>=targets[name]} for name,key in metrics.items()}
 latency=summary['proposal_readiness_seconds']['p95']
 result['proposal_readiness_p95_seconds_max']={'target':120,'observed':latency,'passed':latency is not None and latency<=120}
 gates[split]=result
calibration=read('validation-scored-v1/calibration-result.json')
workload=read('mixed-workload-v1/result.json')
record={'version':'final-candidate-review-v1','frozen_at':datetime.now(timezone.utc).isoformat(),
 'status':'not_release_ready','source_state':'uncommitted_unpushed',
 'base_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),
 'prototype_scope':'public/synthetic; MOCK PMIS; owner sample not expected; owner-authorized AI-assisted review, not independent human adjudication',
 'gates':gates,
 'development_checks':{name:read(name+'/summary.json') for name in ('broad-quality-known-v1','broad-quality-wording-v1','delivery-matching-v1')},
 'calibration':{'status':'experimental_study_only_runtime_unavailable','thresholds':{k:v['review_priority_threshold'] for k,v in calibration['metrics'].items()},'runtime_thresholds_installed':False,'eligible_event_count':7},
 'retry_checks':{name:read(name+'/summary.json') for name in ('broad-quality-wording-retry-v2','delivery-matching-retry-v2') if (base/name/'summary.json').exists()},
 'mixed_workload_retry':read('mixed-workload-retry-v2/result.json'),
 'retained_rehearsal_work':read('mixed-workload-retry-v2/final-inspection.json'),
 'remaining_retries_disposition':read('retry-harness-stop.json'),
 'configuration_sha256':{'compose.yaml':digest(root/'compose.yaml')},
 'mixed_workload':{'status':workload['status'],'duration_seconds':workload['duration_seconds'],'stop_reasons':workload['stop_reasons'],'stages_admitted':workload['stages_admitted'],'rollback':workload['rollback_applied']},
 'monitoring':{name:read(name+'/result.json') for name in ('staged-monitor-600s-v1','mixed-stage-monitor-300s-v1')},
 'verification':read('final-regression-v1.json'),'runtime':read('final-runtime.json'),
 'not_measured':['Acceptance-to-internal-update sustained p95','Mock acknowledgement sustained p95 without retry','Independent hard-safety accuracy across unseen end-to-end approvals'],
 'historical_interruption_cause':'unknown','original_retrieval_failure_root_cause':'unproven',
 'holdout_status':'consumed_for_evaluation; do not tune this candidate or estimator against it; later tuned candidate needs fresh unseen cases',
 'rollout_flags_enabled':False,'automatic_acceptance':False,
 'source_manifest':{'path':'docs/validation/final-candidate-source-2026-10-17.json','sha256':digest(root/'docs/validation/final-candidate-source-2026-10-17.json')},
 'documentation_sha256':{str(p.relative_to(root)):digest(p) for p in [root/'README.md',root/'docs/release.md',root/'docs/README.md',root/'docs/retrieval-resumption-2026-10-17.md',root/'docs/release-review-2026-10-17.md',root/'docs/confidence-calibration.md',root/'docs/local-run.md',root/'docs/data-splits.md',root/'docs/provider-capacity.md']}}
record['evidence_sha256']={str(p.relative_to(root)):digest(p) for p in sorted(base.rglob('*')) if p.is_file() and p.name not in {'final-release-record.json'}}
with (base/'final-release-record.json').open('x') as stream:stream.write(json.dumps(record,indent=2)+'\n')
print(json.dumps({'status':record['status'],'evidence_files':len(record['evidence_sha256']),'record_sha256':digest(base/'final-release-record.json')}))
