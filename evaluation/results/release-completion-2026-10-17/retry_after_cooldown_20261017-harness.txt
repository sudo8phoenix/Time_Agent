from pathlib import Path
from datetime import datetime,timezone
from uuid import UUID
import json,os,subprocess,sys,time
from app.db.session import session_factory
from app.db.models import HostedProviderAdmission,Job,ConversationTask
from app.conversation.runtime import _finish
base=Path('evaluation/results/release-completion-2026-10-17')
def cooled():
 deadline=time.monotonic()+360
 while True:
  with session_factory() as db:
   state=db.get(HostedProviderAdmission,'groq')
   remaining=max(0,(state.state or {}).get('cooldown',0)-time.time()) if state else 0
  if remaining<=0: return
  if time.monotonic()+remaining>deadline: raise SystemExit('Cooldown exceeds bounded retry window; preserve original failures')
  time.sleep(min(5,remaining))
env=os.environ.copy()
env.update(LLM_PROVIDER='groq',REPORT_MODEL_TIMEOUT_SECONDS='90',CONVERSATION_MODEL_TIMEOUT_SECONDS='90',AGENT_HIERARCHY_ENABLED='false',ORCHESTRATOR_MODEL_ASSISTED='false',PYTHONPATH='backend')
cooled()
fixtures=json.loads((base/'mixed-workload-v1/fixtures.json').read_text())
disposition={'recorded_at':datetime.now(timezone.utc).isoformat(),'reason':'Cancel only unclaimed fixtures after observed cooldown stop, before a separate new staged attempt. Original stop remains a failed attempt.','rows':[]}
with session_factory.begin() as db:
 for record in fixtures:
  row=db.get(Job if record['kind']=='report' else ConversationTask,UUID(record['id']))
  if row.state!='queued' or row.attempts!=0 or row.lease_token is not None: raise RuntimeError('Fixture changed; inspect before new staged attempt')
  disposition['rows'].append({'id':record['id'],'kind':record['kind'],'before':row.state,'attempts':row.attempts,'after':'cancelled'})
  if record['kind']=='conversation': _finish(db,row,state='cancelled',error='REHEARSAL_STOPPED')
  else: row.state='cancelled';row.error_code='REHEARSAL_STOPPED';row.error_message='Unclaimed synthetic fixture cancelled after cooldown stop.'
with (base/'mixed-workload-v1/post-stop-disposition.json').open('x') as stream:stream.write(json.dumps(disposition,indent=2)+'\n')
subprocess.run([sys.executable,'-m','evaluation.mixed_workload','--output-dir',str(base/'mixed-workload-retry-v2'),'--seconds','600'],env=env,check=True)
cooled()
subprocess.run([sys.executable,'-m','evaluation.extraction_quality','--fixtures','evaluation/fixtures/broad-quality-wording-v1.json','--case-ids','B19-03','B19-04','B19-05','--prompt-version','extraction-v18-nullable-source-contract','--output-dir',str(base/'broad-quality-wording-retry-v2')],env=env,check=True)
cooled()
subprocess.run([sys.executable,'-m','evaluation.delivery_matching','--output-dir',str(base/'delivery-matching-retry-v2')],env=env,check=True)
