from pathlib import Path
import hashlib,json,os,signal,subprocess,sys,time,urllib.request
from sqlalchemy import func,select,text
from app.db.models import Job,ConversationTask
from app.db.session import session_factory
from app.settings import get_settings
root=Path.cwd()
manifest_path=root/'docs/validation/final-candidate-source-2026-10-17.json'
manifest=json.loads(manifest_path.read_text())
for name,digest in manifest['sha256'].items():
 if hashlib.sha256((root/name).read_bytes()).hexdigest()!=digest: raise SystemExit('Frozen source hash mismatch: '+name)
s=get_settings()
if s.agent_hierarchy_enabled or s.orchestrator_model_assisted: raise SystemExit('Rollout flags must remain disabled')
with session_factory() as db:
 for model in (Job,ConversationTask):
  if db.scalar(select(func.count()).select_from(model).where(model.state.in_(['queued','running']))): raise SystemExit('Main runtime has pending work; no restart performed')
 migration=db.execute(text('select version_num from alembic_version')).scalar_one()
old=json.loads(Path('/private/tmp/progress-completion-20261016-runtime/processes.json').read_text())
expected={'worker':'ops/run_worker.py','api':'-m uvicorn app.main:app --host 127.0.0.1 --port 8001'}
for role,needle in expected.items():
 command=subprocess.check_output(['ps','-p',str(old[role]['pid']),'-o','command='],text=True).strip()
 if needle not in command: raise SystemExit('Existing process identity mismatch: '+role)
for role in ('worker','api'):
 pid=old[role]['pid'];os.kill(pid,signal.SIGTERM)
 deadline=time.monotonic()+15
 while time.monotonic()<deadline:
  found=subprocess.run(['ps','-p',str(pid),'-o','stat='],capture_output=True,text=True)
  if found.returncode or not found.stdout.strip() or found.stdout.strip().startswith('Z'): break
  time.sleep(.2)
 else: raise SystemExit('Existing process did not stop: '+role)
runtime=Path('/private/tmp/progress-final-20261017-runtime');runtime.mkdir(exist_ok=False)
env=os.environ.copy();env['PYTHONPATH']=str(root/'backend')
env['AGENT_HIERARCHY_ENABLED']='false';env['ORCHESTRATOR_MODEL_ASSISTED']='false'
record={'source_manifest_sha256':hashlib.sha256(manifest_path.read_bytes()).hexdigest(),'migration':migration,
 'configuration':{k:getattr(s,k) for k in ['llm_provider','groq_model','report_model_timeout_seconds','conversation_model_timeout_seconds','agent_hierarchy_enabled','orchestrator_model_assisted']},'old_pids':{k:v['pid'] for k,v in old.items()}}
for role,command in [('api',[sys.executable,'-m','uvicorn','app.main:app','--host','127.0.0.1','--port','8001']),('worker',[sys.executable,'ops/run_worker.py'])]:
 with (runtime/(role+'.log')).open('w') as stream:
  process=subprocess.Popen(command,cwd=root,env=env,stdout=stream,stderr=stream,start_new_session=True)
 record[role]={'pid':process.pid,'log':str(runtime/(role+'.log'))}
 (runtime/'processes.json').write_text(json.dumps(record,indent=2)+'\n')
deadline=time.monotonic()+20
while time.monotonic()<deadline:
 try:
  with urllib.request.urlopen('http://127.0.0.1:8001/api/v1/health/live',timeout=2) as response:
   record['health']={'status':response.status,'body':json.loads(response.read())}
  break
 except Exception: time.sleep(.5)
else: raise SystemExit('Final-source API failed health check; inspect preserved runtime logs')
record['source_manifest_verified_before_restart']=True
record['configuration_note']='Existing main report/conversation timeouts preserved; bounded evaluation used 90-second timeouts.'
(runtime/'processes.json').write_text(json.dumps(record,indent=2)+'\n')
(root/'evaluation/results/release-completion-2026-10-17/final-runtime.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record))
